import { registrableDomain } from '../url/domain.js'
 
/**
 * Where the masked values come from (spec 4).
 *
 * ## Never random per call
 *
 * A canvas that returns slightly different pixels every time it is read is not
 * protected, it is *marked*: a site reads the same drawing twice, sees two
 * hashes, and has learned something no ordinary browser does. Every value this
 * module produces is a pure function of a seed, so the same measurement always
 * yields the same answer, however often it is taken.
 *
 * ## Why the seed is per site and not per profile
 *
 * One seed for the whole profile would make the noise itself a cross-site
 * identifier: two unrelated sites could compare the canvas hash they each got
 * and recognise the same user. Deriving the seed from the profile secret *and*
 * the site gives both properties at once — stable for a site, unrelated between
 * sites. This is the same reasoning behind Brave's per-eTLD+1 farbling.
 *
 * The site is the registrable domain, matching how the rest of the browser
 * decides what counts as one site, so `www.example.com` and `cdn.example.com`
 * agree with `example.com` rather than looking like three visitors.
 *
 * ## Why the profile secret is not persisted
 *
 * A seed that survived restarts would be a permanent local identifier: the
 * canvas hash it produces would be stable for years and unique to the install,
 * which is precisely the thing being defended against. It lives for one run of
 * the browser instead — long enough that nothing a site can measure ever
 * disagrees with itself, short enough that it links nothing across sessions.
 */
 
/**
 * FNV-1a, 32-bit.
 *
 * Not a cryptographic hash and does not need to be: nothing about the seed is
 * secret from the machine it runs on, and the property required is only that
 * similar inputs land far apart. A real digest would mean pulling a crypto
 * dependency into the preload for no gain.
 */
export function hash32(text: string): number {
  let hash = 0x811c9dc5
  for (let index = 0; index < text.length; index++) {
    hash ^= text.charCodeAt(index)
    // The FNV prime, as shifts, because `hash * 16777619` loses precision once
    // the product passes 2^53 and would make the result platform-dependent.
    hash = (hash + (hash << 1) + (hash << 4) + (hash << 7) + (hash << 8) + (hash << 24)) >>> 0
  }
  return hash >>> 0
}
 
/**
 * The seed for one site: the profile secret mixed with the registrable domain.
 *
 * The separator matters. Without it, profile `ab` on site `c` and profile `a` on
 * site `bc` would derive the same seed, and a site could confirm a guess about
 * the secret.
 */
export function siteSeed(profileSecret: string, host: string): number {
  return hash32(`${profileSecret}${registrableDomain(host)}`)
}
 
/**
 * xorshift32. Deterministic, seeded, and small enough to be repeated inside the
 * injected script if it ever needs to be.
 */
function nextState(state: number): number {
  let value = state
  value ^= value << 13
  value >>>= 0
  value ^= value >>> 17
  value ^= value << 5
  return value >>> 0
}
 
/** A seed of 0 is a fixed point for xorshift, so it is moved off it. */
function startState(seed: number): number {
  return seed === 0 ? 0x9e3779b9 : seed >>> 0
}
 
/**
 * `count` integers in `[-magnitude, magnitude]`, derived from `seed`.
 *
 * A table rather than a value per call: the injected script must not carry a
 * generator, and a fixed table indexed by position is both cheaper and easier to
 * reason about — the same position always gets the same delta.
 */
export function noiseTable(seed: number, count: number, magnitude: number): number[] {
  const span = magnitude * 2 + 1
  const table: number[] = []
  let state = startState(seed)
  for (let index = 0; index < count; index++) {
    state = nextState(state)
    table.push((state % span) - magnitude)
  }
  return table
}
 
/**
 * An integer in `[0, modulus)` derived from `seed`, used to shift where in a
 * canvas the perturbed pixels fall. Without it every profile would touch the
 * same pixel positions, and the positions themselves would be the fingerprint.
 */
export function seededIndex(seed: number, modulus: number): number {
  return nextState(startState(seed)) % modulus
}
